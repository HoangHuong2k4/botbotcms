GET /api/dealer/stock

Xem tồn kho, giá và số dư hiện tại. Gọi API này để hiển thị menu sản phẩm cho khách.

📤 Request

GET https://api.lay2fa.xyz/api/dealer/stock
X-API-KEY: DLR_xxxxx

📥 Response 200 — Thành công

{
  "success": true,
  "dealer": "Nguyễn Văn A",
  "balance": 500000,
  "products": {
    "gptplus_1thang_KBH": {
      "name": "GPTPlus cá nhân (BH7D)",
      "price": 7000,
      "stock": 50,
      "type": "file"
    },
    "cc1thang": {
      "name": "CapcutPro 35 ngày",
      "price": 15000,
      "stock": 37,
      "type": "api_external"
    }
  }
}
Ý nghĩa field type:
• file: hệ thống giao hàng tự động
• api_external: hệ thống giao hàng tự động, response vẫn trả items như bình thường
Bạn chỉ cần đọc name, price, stock để hiển thị/bán. Không cần xử lý thêm phía sau.

💻 Code Python mẫu

import requests

API_URL = "https://api.lay2fa.xyz"
API_KEY = "DLR_xxxxx"   # ← Thay bằng key thật

def get_stock():
    """Lấy danh sách sản phẩm + stock + giá."""
    try:
        r = requests.get(
            f"{API_URL}/api/dealer/stock",
            headers={"X-API-KEY": API_KEY},
            timeout=10  # ← QUAN TRỌNG: luôn đặt timeout
        )
        data = r.json()

        if data.get("success"):
            balance = data["balance"]
            products = data["products"]
            print(f"Số dư: {balance:,}đ")
            for key, info in products.items():
                print(f"  {key}: {info['name']} | Giá: {info['price']:,}đ | Stock: {info['stock']}")
            return products
        else:
            print(f"Lỗi: {data.get('error')}")
            return None

    except requests.exceptions.ConnectionError:
        print("❌ Không kết nối được server! Kiểm tra lại URL hoặc server đang down")
        return None
    except requests.exceptions.Timeout:
        print("❌ Server không phản hồi (timeout). Thử lại sau")
        return None
    except Exception as e:
        print(f"❌ Lỗi không xác định: {e}")
        return None
💡 Tip: Nên cache kết quả stock trong 30-60 giây, không cần gọi API mỗi lần khách xem menu.

POST /api/dealer/buy

Mua hàng — tự động trừ số dư, trả items. Đây là API chính.

📌 Body Parameters (JSON)

ParamTypeMô tả

💻 Code Python mẫu — Mua sản phẩm thường

def buy_product(product_key, qty):
    """Mua sản phẩm thường hoặc hàng auto xuất. Cùng một format response."""
    try:
        r = requests.post(
            f"{API_URL}/api/dealer/buy",
            headers={
                "X-API-KEY": API_KEY,
                "Content-Type": "application/json"   # ← BẮT BUỘC
            },
            json={
                "product_key": product_key,
                "qty": qty
            },
            timeout=30
        )
        data = r.json()

        if data.get("success"):
            items = data["items"]          # ← Danh sách tài khoản / item bàn giao
            charged = data["total_charged"]
            remaining = data["balance_remaining"]
            order_code = data["order_code"]
            print(f"✅ Mua thành công {qty}x {product_key}")
            print(f"   Trừ: {charged:,}đ | Còn: {remaining:,}đ")
            print(f"   Mã đơn: {order_code}")
            for item in items:
                print(f"   → {item}")
            return items
        else:
            error = data.get("error", "Lỗi không xác định")
            print(f"❌ Mua thất bại: {error}")
            return None

    except requests.exceptions.ConnectionError:
        print("❌ Không kết nối được server!")
        return None
    except requests.exceptions.Timeout:
        print("❌ Timeout — server không phản hồi")
        return None
    except Exception as e:
        print(f"❌ Lỗi: {e}")
        return None

# Sử dụng:
items = buy_product("gptplus_1thang_KBH", 2)
if items:
    # Gửi items cho khách hàng của bạn
    for item in items:
        print(f"Gửi cho khách: {item}")

💻 Code xử lý lỗi mẫu

def safe_buy(product_key, qty):
    """Mua hàng với xử lý lỗi đầy đủ - xử lý đúng HTTP 401/403/404/409/503."""
    try:
        r = requests.post(
            f"{API_URL}/api/dealer/buy",
            headers={
                "X-API-KEY": API_KEY,
                "Content-Type": "application/json"
            },
            json={"product_key": product_key, "qty": qty},
            timeout=60
        )

        # Cac status code rieng - khong can parse JSON
        if r.status_code == 401:
            return None, "Thiếu API key! Kiểm tra header X-API-KEY"
        if r.status_code == 403:
            return None, "API key sai! Gõ /ctv lấy key mới"
        if r.status_code == 404:
            return None, f"Sản phẩm '{product_key}' không tồn tại trong stock"
        if r.status_code == 503:
            return None, f"Sản phẩm '{product_key}' đang tạm ngưng - admin sẽ bật lại sau"

        # 409 (don trung / het hang) va 200 (het tien...) - body JSON van co
        try:
            data = r.json()
        except ValueError:
            return None, f"Server trả về data không hợp lệ (HTTP {r.status_code})"

        if data.get("success"):
            return data["items"], None   # ← Thành công

        # success = False - lay error string
        err = data.get("error", "Lỗi không rõ")
        if r.status_code == 409 and "Khong du hang" in err:
            return None, "Hết hàng, thử qty nhỏ hơn hoặc chờ admin nhập kho"
        if r.status_code == 409:
            return None, "Đơn này đã xử lý rồi (trùng order_code) - đổi mã đơn"
        return None, err

    except requests.exceptions.ConnectionError:
        return None, "Không kết nối được server (server down hoặc sai URL)"
    except requests.exceptions.Timeout:
        return None, "Server không phản hồi (timeout)"
    except Exception as e:
        return None, f"Lỗi: {str(e)}"

# Sử dụng:
items, error = safe_buy("gptplus_1thang_KBH", 1)
if error:
    print(f"❌ {error}")
    # → Thông báo lỗi cho khách
else:
    print(f"✅ Mua thành công!")
    for item in items:
        print(f"  → {item}")
    # → Gửi items cho khách

🤖 Code mẫu bot Telegram hoàn chỉnh

Ví dụ tích hợp vào bot Telegram Python (python-telegram-bot).

import requests
from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup
from telegram.ext import ApplicationBuilder, CommandHandler, CallbackQueryHandler, ContextTypes

# ============================
# CẤU HÌNH — THAY ĐỔI Ở ĐÂY
# ============================
BOT_TOKEN = "YOUR_BOT_TOKEN"
API_URL = "https://api.lay2fa.xyz"
API_KEY = "DLR_xxxxx"  # ← Key từ /ctv

HEADERS = {
    "X-API-KEY": API_KEY,
    "Content-Type": "application/json"
}

# ============================
# HÀM GỌI API
# ============================
def api_get_stock():
    """Lấy danh sách sản phẩm từ API đối tác."""
    try:
        r = requests.get(
            f"{API_URL}/api/dealer/stock",
            headers={"X-API-KEY": API_KEY},
            timeout=10
        )
        data = r.json()
        if data.get("success"):
            return data["products"], data["balance"]
        return None, 0
    except Exception as e:
        print(f"ERROR - Lỗi lấy stock API đối tác: {e}")
        return None, 0

def api_buy(product_key, qty):
    """Mua hàng từ API đối tác."""
    try:
        r = requests.post(
            f"{API_URL}/api/dealer/buy",
            headers=HEADERS,
            json={"product_key": product_key, "qty": qty},
            timeout=60
        )
        # Server tra HTTP 503 khi san pham bi tam ngung
        if r.status_code == 503:
            return {"success": False, "error": "Sản phẩm tạm ngưng, thử sau"}
        return r.json()
    except requests.exceptions.ConnectionError:
        return {"success": False, "error": "Không kết nối được server đối tác"}
    except requests.exceptions.Timeout:
        return {"success": False, "error": "Timeout - server đối tác không phản hồi"}
    except Exception as e:
        return {"success": False, "error": str(e)}

# ============================
# HANDLER BOT TELEGRAM
# ============================
async def menu(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Hiển thị menu sản phẩm."""
    products, balance = api_get_stock()
    if products is None:
        await update.message.reply_text("❌ Server đối tác đang bận, thử lại sau!")
        return

    buttons = []
    for key, info in products.items():
        stock = info["stock"]
        if stock == 0:
            status = "❌ Hết hàng"
        elif stock == -1:
            status = "🔄 Đang cập nhật"
        else:
            status = f"✅ Còn {stock}"

        buttons.append([InlineKeyboardButton(
            f"{info['name']} | {info['price']:,}đ | {status}",
            callback_data=f"buy_{key}"
        )])

    await update.message.reply_text(
        f"🛒 *Menu sản phẩm*\n💰 Số dư: *{balance:,}đ*",
        parse_mode="Markdown",
        reply_markup=InlineKeyboardMarkup(buttons)
    )

async def handle_buy(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Xử lý khi khách nhấn mua."""
    query = update.callback_query
    await query.answer()

    product_key = query.data.replace("buy_", "")

    # Mua 1 sản phẩm
    result = api_buy(product_key, 1)

    if result.get("success"):
        items = result["items"]
        charged = result["total_charged"]
        items_text = "\n".join([f"• `{item}`" for item in items])
        await query.edit_message_text(
            f"✅ *Mua thành công!*\n\n"
            f"📦 Sản phẩm:\n{items_text}\n\n"
            f"💰 Đã trừ: *{charged:,}đ*",
            parse_mode="Markdown"
        )
    else:
        await query.edit_message_text(
            f"❌ *Lỗi:* {result.get('error', 'Không rõ')}"
        )

# ============================
# CHẠY BOT
# ============================
app = ApplicationBuilder().token(BOT_TOKEN).build()
app.add_handler(CommandHandler("menu", menu))
app.add_handler(CallbackQueryHandler(handle_buy, pattern="^buy_"))
app.run_polling()

❓ Câu hỏi thường gặp / Xử lý sự cố

🔴 "Không kết nối được server" / ConnectionError

Nguyên nhân: Server đang down hoặc firewall chặn port 5000.
Cách kiểm tra: Mở trình duyệt → vào https://api.lay2fa.xyz/api/dealer/stock (sẽ trả lỗi 401 nhưng nghĩa là server đang chạy).
Xử lý: Liên hệ admin (Telegram: @longdt8386) để kiểm tra server.

🔴 "Body JSON không hợp lệ" khi POST /buy

Nguyên nhân: Thiếu header Content-Type: application/json.
Xử lý: Luôn gửi kèm header: 

headers = {
    "X-API-KEY": "DLR_xxxxx",
    "Content-Type": "application/json"   # ← QUAN TRỌNG!
}

🔴 HTTP 409 "Khong du hang. Kho con N" / "Don hang da duoc xu ly roi"

Nguyên nhân: Hết kho khi mua, hoặc order_code đã được dùng trước đó.
Xử lý: Bot của bạn nên check r.status_code == 409 riêng để hiển thị thông báo phù hợp cho khách. Nếu là trùng order_code → đổi mã đơn hoặc bỏ trống cho hệ thống tự gen.

🔴 HTTP 503 "San pham 'X' dang tam ngung"

Nguyên nhân: Admin đang tạm tắt sản phẩm này (bảo trì, hết nguồn, v.v.).
Xử lý: Bot bạn nên ẩn sản phẩm khỏi menu đến khi gọi /api/dealer/stock thấy nó xuất hiện trở lại. Sản phẩm tạm ngưng cũng không xuất hiện trong response stock.
